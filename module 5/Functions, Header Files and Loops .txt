// ============================================================================
// CIS 25 - Module 5: Functions, Header Files and Loops
// menu.h - declarations only (the "interface").
//
// Header files store declarations so code can be split across multiple files.
// The include guard below prevents this header from being pasted in twice.
// ============================================================================
#ifndef MENU_H
#define MENU_H

// Prints the menu of options to the screen.
void showMenu();

// Option 1: reads N and prints the running sum 1 + 2 + ... + N (uses a for loop).
void runSum();

// Option 2: reads N and prints its multiplication table 1..12 (uses a for loop).
void runTimesTable();

// Option 3: reads N and counts down N, N-1, ... , 1, "Liftoff!" (uses a while loop).
void runCountdown();

#endif // MENU_H

-------------------- CODE: menu.cpp --------------------


// CIS 25 - Module 5: Functions, Header Files and Loops
// menu.cpp - definitions (the "implementation" of everything menu.h declares).

#include "menu.h"
#include <iostream>

void showMenu() {
    // One streamed write keeps the menu fast and flicker-free.
    std::cout << "\n===== FUNCTION MENU =====\n"
                 "1. Sum from 1 to N\n"
                 "2. Multiplication table of N\n"
                 "3. Countdown from N\n"
                 "4. Exit\n"
                 "Choose an option: ";
}

void runSum() {
    int n;
    std::cout << "Enter N: ";
    std::cin >> n;

    long long total = 0;                       // long long guards against overflow
    for (int i = 1; i <= n; ++i) {             // for loop: count is known
        total += i;
    }
    std::cout << "Sum from 1 to " << n << " = " << total << '\n';
}

void runTimesTable() {
    int n;
    std::cout << "Enter N: ";
    std::cin >> n;

    for (int i = 1; i <= 12; ++i) {            // for loop over a fixed range
        std::cout << n << " x " << i << " = " << (n * i) << '\n';
    }
}

void runCountdown() {
    int n;
    std::cout << "Enter N: ";
    std::cin >> n;

    while (n > 0) {                            // while loop: run while condition true
        std::cout << n << ' ';
        --n;
    }
    std::cout << "Liftoff!\n";
}

-------------------- CODE: main.cpp --------------------

// ============================================================================
// CIS 25 - Module 5: Functions, Header Files and Loops
// main.cpp - drives the program.
//
// Requirements met:
//   * A menu with at least three options.                 (four options below)
//   * Each menu option calls a function that does something.
//   * Header files are included (our own "menu.h" + standard <iostream>).
//   * A loop keeps the menu running until the user chooses Exit.
//
// Build: g++ -std=c++17 -Wall main.cpp menu.cpp -o module5
// ============================================================================
#include "menu.h"
#include <iostream>

int main() {
    std::ios::sync_with_stdio(false);

    int option = 0;
    do {                                       // do-while: menu shows at least once
        showMenu();
        std::cin >> option;

        // Dispatch to the function for the chosen option.
        if (option == 1) {
            runSum();
        } else if (option == 2) {
            runTimesTable();
        } else if (option == 3) {
            runCountdown();
        } else if (option == 4) {
            std::cout << "Goodbye!\n";
        } else {
            std::cout << "Invalid option. Please choose 1-4.\n";
        }
    } while (option != 4);                      // loop until Exit is selected

    return 0;
}
