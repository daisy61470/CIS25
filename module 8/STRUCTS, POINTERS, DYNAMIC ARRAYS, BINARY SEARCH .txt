// ============================================================================
// MODULE 8: C++ BASICS - STRUCTS, POINTERS, DYNAMIC ARRAYS, BINARY SEARCH
// Course: CIS 25
// Name:   Daisy Schall
//
// Hospital Patient Management System
//
// Menu options required by the assignment:
//   1. Add a patient
//   2. Display all patients
//   3. Search for a patient by ID (Binary Search)
//   4. Update patient information
//   5. Save patients to file
//   6. Exit program  -> prints "Good Bye!" and releases memory
//
// On startup the program loads patients from patients.txt when that file
// already exists.
//
// DATA FILE STRUCTURE (patients.txt)
//   One patient per line, fields separated by the pipe character '|'
//   id|name|age|gender|diagnosis
//   The pipe is used instead of a comma so names and diagnoses may hold spaces.
//
// Compile: g++ -std=c++17 -Wall module8.cpp -o module8
// Run:     ./module8
// ============================================================================

#include <iostream>
#include <fstream>    // ifstream / ofstream for the patients.txt file
#include <sstream>    // istringstream for splitting a saved line
#include <string>
#include <limits>     // numeric_limits, used to clear bad menu input
#include <iomanip>    // setw / left for the display table
using namespace std;

// ----------------------------------------------------------------------------
// STEP 1: struct that stores one patient record
// ----------------------------------------------------------------------------
struct Patient {
    int    id;
    string name;
    int    age;
    string gender;
    string diagnosis;
};

// ----------------------------------------------------------------------------
// Program constants
// ----------------------------------------------------------------------------
const int    CAPACITY  = 100;              // maximum patients the array can hold
const string DATA_FILE = "patients.txt";   // external file used for persistence

// ----------------------------------------------------------------------------
// Keeps the array ordered by ascending id.
// Binary search only works on sorted data, so this runs after every add,
// after every load, and after any id change during an update.
// Insertion sort is used since the record count stays small.
// ----------------------------------------------------------------------------
void sortByID(Patient* patients, int count) {
    for (int i = 1; i < count; i++) {
        Patient key = patients[i];          // record being placed
        int j = i - 1;
        while (j >= 0 && patients[j].id > key.id) {
            patients[j + 1] = patients[j];  // shift larger ids one slot right
            j--;
        }
        patients[j + 1] = key;
    }
}

// ----------------------------------------------------------------------------
// STEP 6: Binary search on the sorted dynamic array.
// Returns the index of the matching patient, or -1 when no match exists.
// Each pass halves the search window, giving O(log n) instead of O(n).
// ----------------------------------------------------------------------------
int binarySearch(Patient* patients, int size, int searchID) {
    int left  = 0;
    int right = size - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;   // midpoint, overflow safe

        if (patients[mid].id == searchID) {
            return mid;                        // found it
        }
        if (patients[mid].id < searchID) {
            left = mid + 1;                    // target sits in the right half
        } else {
            right = mid - 1;                   // target sits in the left half
        }
    }
    return -1;                                 // window collapsed, no match
}

// ----------------------------------------------------------------------------
// Prints one patient as an aligned table row.
// ----------------------------------------------------------------------------
void printRow(const Patient& p) {
    cout << left << setw(8)  << p.id
         << left << setw(18) << p.name
         << left << setw(7)  << p.age
         << left << setw(10) << p.gender
         << left << setw(24) << p.diagnosis
         << endl;
}

// ----------------------------------------------------------------------------
// STEP 4: Add a patient into the next free slot of the dynamic array.
// ----------------------------------------------------------------------------
void addPatient(Patient* patients, int& count) {
    if (count >= CAPACITY) {
        cout << "\nThe system is full. No more patients can be added.\n";
        return;
    }

    Patient p;

    cout << "\n--- Add a Patient ---\n";
    cout << "Enter Patient ID: ";
    cin  >> p.id;

    // Reject a duplicate id, since binary search assumes unique keys.
    if (binarySearch(patients, count, p.id) != -1) {
        cout << "A patient with ID " << p.id << " already exists.\n";
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        return;
    }

    cin.ignore(numeric_limits<streamsize>::max(), '\n');  // drop the leftover newline
    cout << "Enter Name: ";
    getline(cin, p.name);

    cout << "Enter Age: ";
    cin  >> p.age;
    cin.ignore(numeric_limits<streamsize>::max(), '\n');

    cout << "Enter Gender: ";
    getline(cin, p.gender);

    cout << "Enter Diagnosis: ";
    getline(cin, p.diagnosis);

    patients[count] = p;      // store at the end of the used region
    count++;                  // one more record now lives in the array
    sortByID(patients, count);

    cout << "\nPatient added successfully!\n";
}

// ----------------------------------------------------------------------------
// STEP 5: Display every stored patient.
// ----------------------------------------------------------------------------
void displayAllPatients(const Patient* patients, int count) {
    cout << "\n--- All Patient Records ---\n";
    if (count == 0) {
        cout << "No patients on record.\n";
        return;
    }

    cout << string(67, '-') << endl;
    cout << left << setw(8)  << "ID"
         << left << setw(18) << "NAME"
         << left << setw(7)  << "AGE"
         << left << setw(10) << "GENDER"
         << left << setw(24) << "DIAGNOSIS" << endl;
    cout << string(67, '-') << endl;

    for (int i = 0; i < count; i++) {
        printRow(patients[i]);
    }
    cout << string(67, '-') << endl;
    cout << "Total patients: " << count << endl;
}

// ----------------------------------------------------------------------------
// STEP 6 wrapper: ask for an id, run the binary search, report the result.
// ----------------------------------------------------------------------------
void searchPatient(Patient* patients, int count) {
    cout << "\n--- Search Patient by ID (Binary Search) ---\n";
    if (count == 0) {
        cout << "No patients on record.\n";
        return;
    }

    int searchID;
    cout << "Enter Patient ID to search: ";
    cin  >> searchID;
    cin.ignore(numeric_limits<streamsize>::max(), '\n');

    int result = binarySearch(patients, count, searchID);

    if (result != -1) {
        cout << "\nPatient Found\n";
        cout << "ID:        " << patients[result].id        << endl;
        cout << "Name:      " << patients[result].name      << endl;
        cout << "Age:       " << patients[result].age       << endl;
        cout << "Gender:    " << patients[result].gender    << endl;
        cout << "Diagnosis: " << patients[result].diagnosis << endl;
    } else {
        cout << "\nPatient not found\n";
    }
}

// ----------------------------------------------------------------------------
// STEP 7: Locate a patient with binary search, then overwrite their fields.
// Pressing Enter on a prompt keeps the existing value.
// ----------------------------------------------------------------------------
void updatePatient(Patient* patients, int count) {
    cout << "\n--- Update Patient Information ---\n";
    if (count == 0) {
        cout << "No patients on record.\n";
        return;
    }

    int searchID;
    cout << "Enter Patient ID to update: ";
    cin  >> searchID;
    cin.ignore(numeric_limits<streamsize>::max(), '\n');

    int index = binarySearch(patients, count, searchID);
    if (index == -1) {
        cout << "\nPatient not found\n";
        return;
    }

    cout << "\nCurrent record:\n";
    printRow(patients[index]);

    string buffer;
    cout << "\nEnter new Name (press Enter to keep \""
         << patients[index].name << "\"): ";
    getline(cin, buffer);
    if (!buffer.empty()) patients[index].name = buffer;

    cout << "Enter new Age (press Enter to keep " << patients[index].age << "): ";
    getline(cin, buffer);
    if (!buffer.empty()) patients[index].age = stoi(buffer);

    cout << "Enter new Gender (press Enter to keep \""
         << patients[index].gender << "\"): ";
    getline(cin, buffer);
    if (!buffer.empty()) patients[index].gender = buffer;

    cout << "Enter new Diagnosis (press Enter to keep \""
         << patients[index].diagnosis << "\"): ";
    getline(cin, buffer);
    if (!buffer.empty()) patients[index].diagnosis = buffer;

    cout << "\nPatient record updated successfully!\n";
}

// ----------------------------------------------------------------------------
// STEP 8: Write every patient to patients.txt using the pipe format.
// ----------------------------------------------------------------------------
void savePatients(const Patient* patients, int count) {
    ofstream outFile(DATA_FILE);
    if (!outFile) {
        cout << "\nERROR: could not open " << DATA_FILE << " for writing.\n";
        return;
    }

    for (int i = 0; i < count; i++) {
        outFile << patients[i].id        << '|'
                << patients[i].name      << '|'
                << patients[i].age       << '|'
                << patients[i].gender    << '|'
                << patients[i].diagnosis << '\n';
    }

    outFile.close();   // flush and release the handle
    cout << "\n" << count << " patient record(s) saved to " << DATA_FILE << "\n";
}

// ----------------------------------------------------------------------------
// Startup load. A missing file is normal on a first run, so no error is shown.
// ----------------------------------------------------------------------------
void loadPatients(Patient* patients, int& count) {
    ifstream inFile(DATA_FILE);
    if (!inFile) {
        cout << "No existing data file found. Starting with an empty system.\n";
        return;
    }

    string line;
    while (count < CAPACITY && getline(inFile, line)) {
        if (line.empty()) continue;

        istringstream ss(line);
        string idText, ageText;
        Patient p;

        // Split the five pipe-delimited fields back into the struct.
        getline(ss, idText,      '|');
        getline(ss, p.name,      '|');
        getline(ss, ageText,     '|');
        getline(ss, p.gender,    '|');
        getline(ss, p.diagnosis, '|');

        if (idText.empty() || ageText.empty()) continue;   // skip malformed lines

        p.id  = stoi(idText);
        p.age = stoi(ageText);

        patients[count] = p;
        count++;
    }

    inFile.close();
    sortByID(patients, count);   // guarantee the array is search-ready
    cout << count << " patient record(s) loaded from " << DATA_FILE << "\n";
}

// ----------------------------------------------------------------------------
// STEP 3: The menu
// ----------------------------------------------------------------------------
void showMenu() {
    cout << "\n========================================\n";
    cout << "   HOSPITAL PATIENT MANAGEMENT SYSTEM\n";
    cout << "========================================\n";
    cout << "1. Add a patient\n";
    cout << "2. Display all patients\n";
    cout << "3. Search for a patient by ID\n";
    cout << "4. Update patient information\n";
    cout << "5. Save patients to file\n";
    cout << "6. Exit program\n";
    cout << "----------------------------------------\n";
    cout << "Enter your choice (1-6): ";
}

// ----------------------------------------------------------------------------
// MAIN
// ----------------------------------------------------------------------------
int main() {
    // STEP 2: dynamic array on the heap, reached through a Patient pointer.
    Patient* patients = new Patient[CAPACITY];
    int count = 0;                       // number of slots currently in use

    cout << "Starting Hospital Patient Management System...\n";
    loadPatients(patients, count);       // pull in saved records if they exist

    int choice = 0;
    do {
        showMenu();

        // Guard against letters or symbols typed at the menu prompt.
        if (!(cin >> choice)) {
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
            cout << "\nInvalid input. Please enter a number from 1 to 6.\n";
            continue;
        }
        cin.ignore(numeric_limits<streamsize>::max(), '\n');

        switch (choice) {
            case 1: addPatient(patients, count);        break;
            case 2: displayAllPatients(patients, count); break;
            case 3: searchPatient(patients, count);      break;
            case 4: updatePatient(patients, count);      break;
            case 5: savePatients(patients, count);       break;
            case 6: cout << "\nGood Bye!\n";             break;
            default:
                cout << "\nInvalid choice. Please enter a number from 1 to 6.\n";
        }
    } while (choice != 6);

    // Clean shutdown: hand the heap memory back and drop the dangling pointer.
    delete[] patients;
    patients = nullptr;

    return 0;
}
